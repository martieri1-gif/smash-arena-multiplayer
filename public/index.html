const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const path = require('path'); const app = express(); const server = http.createServer(app); const io = new Server(server); const PORT = process.env.PORT || 3000; const PUBLIC_DIR = path.join(__dirname, 'public'); app.use(express.static(PUBLIC_DIR)); app.get('/health', (_req, res) => { res.json({ ok: true, service: 'smash-arena' }); }); const rooms = new Map(); function makeCode() { const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let code = ''; for (let i = 0; i < 5; i++) code += chars[Math.floor(Math.random() * chars.length)]; return code; } function createRoomCode() { let code = makeCode(); while (rooms.has(code)) code = makeCode(); return code; } function getRoom(code) { return rooms.get(code.toUpperCase()); } function ensurePlayerEntry(player) { return { id: player.id, name: player.name || 'Player', charId: player.charId || 'ninja', roomCode: player.roomCode || null, x: 180, y: 0, vx: 0, vy: 0, w: 48, h: 48, damage: 0, facing: 1, stocks: 3, onGround: false, attackCooldown: 0, specialCooldown: 0, hitStun: 0, isDead: false, input: { left: false, right: false, jump: false, attack: false, special: false, } }; } function resetPlayer(player, isLeft) { player.x = isLeft ? 180 : 980; player.y = 0; player.vx = 0; player.vy = 0; player.damage = 0; player.facing = isLeft ? 1 : -1; player.hitStun = 0; player.onGround = false; player.isDead = false; player.attackCooldown = 0; player.specialCooldown = 0; } function setPlayerInput(player, input) { const normalized = { left: !!input.left, right: !!input.right, jump: !!input.jump, attack: !!input.attack, special: !!input.special, }; player.input = normalized; } function roomState(room) { return { code: room.code, players: room.players.map((p) => ({ id: p.id, name: p.name, charId: p.charId, x: p.x, y: p.y, w: p.w, h: p.h, damage: p.damage, facing: p.facing, stocks: p.stocks, onGround: p.onGround, hitStun: p.hitStun, attackCooldown: p.attackCooldown, specialCooldown: p.specialCooldown, isDead: p.isDead, })), timer: room.timer, timerMax: room.timerMax, roomReady: room.players.length >= 2, winner: room.winner || null, started: room.started, }; } function broadcastRoom(room) { io.to(room.code).emit('roomState', roomState(room)); } function joinRoom(socket, code, name, charId) { code = (code || '').toUpperCase(); const existing = getRoom(code); if (!existing) { socket.emit('roomError', 'Room not found'); return; } if (existing.players.length >= 2) { socket.emit('roomError', 'Room is full'); return; } const player = ensurePlayerEntry({ id: socket.id, name, charId, roomCode: code, }); existing.players.push(player); socket.join(code); socket.data.roomCode = code; socket.data.playerId = socket.id; if (existing.players.length === 2) { existing.started = true; existing.timer = 99; existing.timerMax = 99; existing.winner = null; existing.players.forEach((p, index) => { resetPlayer(p, index === 0 ? true : false); p.charId = index === 0 ? existing.players[0].charId : existing.players[1].charId; }); existing.players[0].name = existing.players[0].name || 'Player 1'; existing.players[1].name = existing.players[1].name || 'Player 2'; io.to(code).emit('matchStart', { code, players: existing.players.map((p) => ({ id: p.id, name: p.name, charId: p.charId })) }); } broadcastRoom(existing); } function createRoom(socket, roomName, charId) { const code = createRoomCode(); const room = { code, players: [], timer: 99, timerMax: 99, started: false, winner: null, }; rooms.set(code, room); socket.join(code); socket.data.roomCode = code; socket.data.playerId = socket.id; const player = ensurePlayerEntry({ id: socket.id, name: roomName || 'Host', charId, roomCode: code, }); room.players.push(player); socket.emit('roomCreated', { code }); broadcastRoom(room); } function processAttack(attacker, target) { if (attacker.attackCooldown > 0 || target.isDead) return; attacker.attackCooldown = 22; const damage = 6 + attacker.damage * 0.1 + (attacker.charId === 'knight' ? 7 : 0); const knockback = 12; if (target.invincible) return; const dx = target.x - attacker.x; const direction = dx >= 0 ? 1 : -1; target.damage = Math.min(999, target.damage + damage); target.vx = direction * knockback; target.vy = -8; target.hitStun = 12; if (target.damage >= 100) { target.stocks -= 1; target.damage = 0; target.isDead = true; target.x = target.x < 600 ? 180 : 980; target.y = 0; target.vx = 0; target.vy = 0; if (target.stocks <= 0) { attacker.roomWinner = true; } } } function handlePlayerMovement(player, room) { if (!player || player.isDead) return; const left = player.input.left; const right = player.input.right; const jump = player.input.jump; const attack = player.input.attack; const special = player.input.special; player.attackCooldown = Math.max(0, player.attackCooldown - 1); player.specialCooldown = Math.max(0, player.specialCooldown - 1); player.hitStun = Math.max(0, player.hitStun - 1); if (!player.hitStun) { if (left) { player.vx = -5; player.facing = -1; } else if (right) { player.vx = 5; player.facing = 1; } else { player.vx *= 0.72; } } if (jump && player.onGround && !player.hitStun) { player.vy = -14; player.onGround = false; } if (attack && player.attackCooldown <= 0 && !player.hitStun) { const other = room.players.find((p) => p.id !== player.id); if (other) processAttack(player, other); } if (special && player.specialCooldown <= 0 && !player.hitStun) { player.specialCooldown = 110; const other = room.players.find((p) => p.id !== player.id); if (other) { other.damage = Math.min(999, other.damage + 18); other.vx = (other.x > player.x ? 1 : -1) * 15; other.vy = -12; } } player.vy += 0.6; player.x += player.vx; player.y += player.vy; if (player.y > 500) { player.y = 0; player.vy = 0; player.onGround = true; player.x = player.x < 600 ? 180 : 980; } if (player.y >= 420) { player.y = 420; player.vy = 0; player.onGround = true; } player.x = Math.max(40, Math.min(1120, player.x)); } function tickRoom(room) { if (!room.started || room.players.length < 2) return; room.timer = Math.max(0, room.timer - 1 / 60); if (room.timer <= 0) { const highest = [...room.players].sort((a, b) => b.damage - a.damage)[0]; room.winner = highest ? highest.name : 'Draw'; room.started = false; io.to(room.code).emit('matchEnd', { winner: room.winner, code: room.code }); } room.players.forEach((player) => handlePlayerMovement(player, room)); room.players.forEach((player) => { if (player.stocks <= 0) { player.isDead = true; } }); const p1 = room.players[0]; const p2 = room.players[1]; if (p1 && p2) { if (p1.stocks <= 0 || p2.stocks <= 0) { const winner = p1.stocks > p2.stocks ? p1 : p2; room.winner = winner.name; room.started = false; io.to(room.code).emit('matchEnd', { winner: room.winner, code: room.code }); } } broadcastRoom(room); } setInterval(() => { for (const room of rooms.values()) { tickRoom(room); } }, 1000 / 60); io.on('connection', (socket) => { socket.emit('welcome', { status: 'connected' }); socket.on('createRoom', ({ name, charId }) => { createRoom(socket, name || 'Host', charId || 'ninja'); }); socket.on('joinRoom', ({ code, name, charId }) => { joinRoom(socket, code, name || 'Player', charId || 'ninja'); }); socket.on('input', (payload) => { const roomCode = socket.data.roomCode; if (!roomCode) return; const room = getRoom(roomCode); if (!room) return; const player = room.players.find((p) => p.id === socket.id); if (!player) return; setPlayerInput(player, payload || {}); }); socket.on('disconnect', () => { const roomCode = socket.data.roomCode; if (!roomCode) return; const room = getRoom(roomCode); if (!room) return; room.players = room.players.filter((p) => p.id !== socket.id); if (room.players.length === 0) { rooms.delete(roomCode); } else { room.started = room.players.length >= 2; broadcastRoom(room); } }); }); server.listen(PORT, () => { console.log(`SMASH ARENA server running on http://localhost:${PORT}`); });
